option(KUNLUN_USE_SYSTEM_LIBRARY "Use an installed KunlunK200 package instead of the submodule" OFF)
if(KUNLUN_USE_SYSTEM_LIBRARY)
    find_package(KunlunK200 0.7.7 CONFIG REQUIRED)
elseif(NOT TARGET KunlunK200::kunlunk200)
    set(kunlun_source "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/KunlunK200")
    if(NOT EXISTS "${kunlun_source}/CMakeLists.txt")
        message(FATAL_ERROR "Initialize vendor/KunlunK200: git submodule update --init --recursive")
    endif()
    add_subdirectory("${kunlun_source}" "${CMAKE_BINARY_DIR}/kunlunk200")
endif()
ggml_add_backend_library(ggml-kunlun ggml-kunlun.cpp)
target_link_libraries(ggml-kunlun PRIVATE KunlunK200::kunlunk200)
